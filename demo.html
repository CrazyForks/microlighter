JavaScript

/* Begin/end regions, captures, and constants */
class ThemeSwitcher extends HTMLElement {
  connectedCallback() {
    const selectedTheme = this.getAttribute("data-syntax-theme") ?? "github";

    if (selectedTheme === "vscode-plus") {
      document.documentElement.dataset.syntaxTheme = selectedTheme;
    }

    return true;
  }
}

customElements.define("theme-switcher", ThemeSwitcher);

Ruby

class SyntaxTheme
  DEFAULT_THEME = :dracula

  def initialize(name = DEFAULT_THEME)
    @name = name
  end

  def label
    "Theme: #{@name}"
  end
end

theme = SyntaxTheme.new
puts theme.label if theme
